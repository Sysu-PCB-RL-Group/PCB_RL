# ========== Reward 权重（负值为惩罚，正值为奖励） ==========
W_OVERLAP_HARD        = 20.0      # 硬重叠惩罚（元件物理碰撞，必须避免）
W_OVERLAP_SOFT        = 400.0     # 软间距惩罚（元件距离太近但不重叠）v3: 100→400，贴边代价反超完成奖励，消除"无脑贴边"reward hacking
W_OUT_OF_BOUNDS_INITIAL= 2.0      # 出板框初始惩罚（curriculum 前期宽松）
W_OUT_OF_BOUNDS_FINAL  = 20.0     # 出板框最终惩罚（curriculum 后期>500ep 收紧至此值）
W_NET_AGGREGATION     = 5.0       # 网络聚集奖励（同一网络元件越近越好）v3: 10→5，削弱成团推手
W_CONGESTION          = 30.0      # 拥塞均衡奖励（引导元件分布均匀）v3: 15→30，配合软间距强化分散
W_DECOUPLING           = 10.0     # 去耦电容配对奖励（去耦电容靠近对应 IC）
W_LARGE_COMP           = 5.0      # 大型元件优先奖励（大元件早放有更多空位可选）
W_COMPLETION           = 100.0    # 完成奖励（每成功放置一颗元件的基础分）

# ========== Reward Shaping：品质项时序 ==========
QUALITY_START_EP      = 500       # 品质项开始介入的 episode（约束学稳后引入品质引导）v3: 3000→500
QUALITY_RAMP_EP       = 1500      # 品质项线性升温 episode 数（到 START+RAMP 时品质权重=1.0）v3: 2000→1500

# ========== 训练参数 ==========
GAMMA               = 0.99       # 折扣因子（0.99^5≈0.95，5步子任务间有效；跨元件衰减大但局部优先有益）
EPSILON_START       = 0.6        # ε-greedy 初始探索率（BC 预训练后无需 1.0 完全随机）
EPSILON_MIN         = 0.1        # ε-greedy 最小探索率（指数衰减下界）
EPSILON_DECAY_EPISODES = 10000  # ε 衰减最短收敛区间（下限）：总 epi ≤ 此值时按此值收敛，> 此值时按实际总 epi 收敛
MAX_STEPS           = 2000       # 单 episode 最大环境步数（233 元件×5 子步≈1165，留余量）
EARLY_STOP_THRESHOLD  = -150.0   # 累计 reward 低于此值提前终止 episode
FUSE_MAX_CONSECUTIVE_FAIL = 10   # 保险丝：同一元件连续失败 ≥ N 次强制终止 episode（打断坏循环重试，0=禁用）
CURRICULUM_START_EP = 0          # curriculum 起始 episode（出界惩罚从轻到重）
CURRICULUM_END_EP   = 500        # curriculum 截止 episode（500ep 后出界惩罚=最终值）
TOTAL_EPISODES      = 10000      # 总训练 episode 数

# ========== DQN 训练参数 ==========
BATCH_SIZE          = 64         # 每次从 replay buffer 采样的批量大小
REPLAY_BUFFER_SIZE  = 100000     # 经验回放池容量（存满后旧经验被覆盖）
LEARNING_RATE       = 0.0001     # Adam 优化器学习率
TARGET_UPDATE_FREQ  = 20         # target 网络同步频率（每 N 个 episode 更新一次目标网络）
WARMUP_EPISODES     = 500        # 预热期：前 N 个 episode 只收集不学习，避免早期噪声样本污染梯度

# ========== 几何阈值（单位：mm） ==========
BOARD_MARGIN        = 0.0        # 板框内缩边距（元件必须在此范围内）
DECOUPLING_MAX_DIST = 5.0        # 去耦电容到 IC 的最大有效距离
CELL_SIZE           = 0.2        # 网格离散化单位（0.2mm/格，细化 5 倍以表达真实元件间距）
CONGESTION_CELL_SIZE= 2.0        # 拥塞检测的单元格大小
DEFAULT_SPACING     = 0.25       # 默认元件间距（不匹配特定封装类型时使用）

# ========== 分类间距阈值（单位：mm，同类封装之间最小距离） ==========
SPACING_CHIP        = 0.2        # 片式元件（电阻电容电感等小封装）
SPACING_SOIC        = 0.2        # SOIC/SOP 等翼形引脚 IC（统一阈值，现阶段无大封装板）
SPACING_QFP         = 0.2        # QFP/LQFP 四面引脚 IC（统一阈值）
SPACING_BGA         = 0.2        # BGA 球栅阵列 IC（统一阈值）
SPACING_IC          = 0.2        # 通用 IC 间距（统一阈值）

# ========== 专家轨迹注入（Expert Trajectory Injection） ==========
EXPERT_TOTAL_TRANSITIONS      = 20000  # 目标专家 transition 总数（生成直到达到即停，略超可接受）
EXPERT_MAX_TRAJECTORIES       = 5000   # 最大轨迹数安全上限（防死循环）
EXPERT_EPSILON_DECOMPOSE      = 0.5    # 拆解随机化概率（50% 随机选 / 50% 选最小面积）
EXPERT_POS_NOISE_PROB         = 0.0    # 位置变异概率（原 0.2）：消除同一 obs 的坐标歧义。诊断发现 20% 位置 ±1~2 格变异让同一 obs 对应多个矛盾坐标，BC 坐标 head 无法精确拟合（top-1 仅 8~15%），改为 0 让专家轨迹坐标目标唯一
EXPERT_ROT_NOISE_PROB         = 0.1    # 旋转变异概率（全 4 种旋转）
EXPERT_BC_EPOCHS              = 150    # BC 预训练 epoch 数（原 50）：BC loss 2.46→1.80 仍在降未收敛，坐标 head 学不会精确位置，加至 150 提升学习容量
EXPERT_APPLY_PENALTIES        = True   # 专家 reward 是否携带惩罚项（True=与 RL 真实环境一致；False=仅品质奖励，实验 D 对照用）
EXPERT_BUFFER_RATIO           = 0.2    # 第二阶段 buffer 中专家经验占比（每 batch 采样占比 20%→70%）
EXPERT_BUFFER_MULTIPLIER      = 5      # 第二阶段 buffer 总容量 = 此值 × 专家 transition 总数
PER_ALPHA                     = 0.6    # 优先经验回放 α（0=均匀，1=完全按优先级）
PER_BETA_START                = 0.4    # 重要性采样 β 初始值（向 1.0 递增）
BC_REG_LAMBDA                 = 10.0   # BC 正则化强度(策略锚定λkl)：RL loss += λkl·KL(当前策略‖冻结BC策略)；贪心甜点定稿=10（T3 15步/+77.8最优，T4=2/T5=5退化）；λv=10仍钉Q防发散
BC_REG_TEMP                   = 1.0    # BC 正则化 softmax 温度（策略蒸馏温度）
BC_REG_VALUE_LAMBDA           = 10.0   # BC 值正则化强度：RL loss += λv·MSE(Q当前, Q_BC)，锚定 Q 值幅度防发散（λv=0.5 太弱压不住 Q 漂）
EXPERT_TRANSITIONS_DIR        = data/expert_transitions  # 专家 transition 序列化存储目录

# ========== 层动作约束 ==========
SINGLE_LAYER        = True       # True=锁定所有元件到原始层（单面板/复刻任务，堵"翻面钻空子"reward hacking）；False=按 footprint 物理层约束（SMD 锁焊盘面，PTH 可双面）
